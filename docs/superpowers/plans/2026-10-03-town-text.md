# TOWN.DAT Text Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Translate the Notebook articles and town UI text in `TOWN.DAT` through the existing JSON → build pipeline.

**Architecture:** New `town.py` parses the nested word-offset containers of a TOWN sub-file into a tree, parses/builds the two text leaves (u16 offset table of `0001 <sjis> 0000` lines), and rebuilds the tree with grown leaves. `build.dump()` writes `script/TOWN/notebook.json` and `town.json` once; `build.patch_town()` rewrites every copy of the leaves in all 28 town-state files. Translation goes through `tl.py` like before.

**Tech Stack:** Python 3.12 stdlib, existing `dat.py`, `script.encode/decode/wrap`, `tl.py`.

**Spec:** `docs/superpowers/specs/2026-10-03-town-text-design.md`

## Global Constraints

- Python stdlib only; tests are plain functions run with `python test_x.py` (and pytest-compatible).
- Byte-identical round trip of every untouched structure; identity build (`test_build.py`) stays byte-identical to the ROM.
- Strings are `script.encode` output (even byte length; odd ASCII runs padded with a space); never write control bytes.
- Line limits: Notebook 32 bytes, town UI 28 bytes. Text leaf < 65536 bytes.
- Never write into bss/zero tails of code; TOWN leaves are data, growth is via container relayout only.
- Commit after each task with `Co-Authored-By: Claude Fable 5.1 <noreply@anthropic.com>`.

## Review Focus

1. A container whose entries are not contiguous or not word-aligned must be treated as a leaf, not rebuilt differently (Task 1 round-trip test over all 43 sub-files).
2. A text entry with zero lines (empty entry, shared offset) must survive parse/build (Task 2 test).
3. An English string with an odd byte count must still be followed by an aligned `0000` terminator (Task 2 test via `encode` padding).
4. A translated Notebook paragraph longer than 32 bytes must wrap, not raise, and a blank paragraph must become a `　` line (Task 3 test).
5. Growth that pushes the parent container's later siblings must keep their bytes identical at new offsets (Task 3 test).

```python
"""TOWN.DAT: nested containers (word offsets) holding two text leaves per town file."""
import struct
import script

Node = "bytes | tuple[list, bytes]"


def _table(b: bytes):
    """(offset, size) byte pairs if b starts with a container header, else None."""
    if len(b) < 8:
        return None
    n, one, ver = struct.unpack_from("<HHI", b, 0)
    if one != 1 or ver != 2 or n == 0 or 8 + 8 * n > len(b):
        return None
    pairs = [(o * 4, s * 4) for o, s in (struct.unpack_from("<II", b, 8 + 8 * i) for i in range(n))]
    pos = 8 + 8 * n
    for o, s in pairs:                       # contiguous, in range
        if o != pos or o + s > len(b):
            return None
        pos += s
    return pairs


def parse(b: bytes):
    """A leaf is bytes; a container is (children, tail) with tail = bytes after the last entry."""
    pairs = _table(b)
    if pairs is None:
        return b
    children = [parse(b[o:o + s]) for o, s in pairs]
    end = pairs[-1][0] + pairs[-1][1]
    return children, b[end:]


def rebuild(node) -> bytes:
    if isinstance(node, bytes):
        return node
    children, tail = node
    blobs = [rebuild(c) for c in children]
    blobs = [x + bytes(-len(x) % 4) for x in blobs]
    out = bytearray(struct.pack("<HHI", len(blobs), 1, 2))
    pos = 8 + 8 * len(blobs)
    for x in blobs:
        out += struct.pack("<II", pos // 4, len(x) // 4)
        pos += len(x)
    for x in blobs:
        out += x
    return bytes(out) + tail
```

```python
def text_entries(leaf: bytes):
    """[[(param, sjis), ...] per entry] or None. Entry = (0001 [param] str 0000)* 0000."""
    if len(leaf) < 4 or len(leaf) % 2:
        return None
    first = struct.unpack_from("<H", leaf, 0)[0]
    if first < 2 or first % 2 or first > len(leaf):
        return None
    offs = [struct.unpack_from("<H", leaf, 2 * i)[0] for i in range(first // 2)] + [len(leaf)]
    if any(a > b for a, b in zip(offs, offs[1:])):
        return None
    entries = []
    for a, b in zip(offs, offs[1:]):
        e, pos, lines = leaf[a:b], 0, []
        while pos < len(e):
            op = struct.unpack_from("<H", e, pos)[0]
            pos += 2
            if op == 0:
                if pos != len(e):
                    return None
                break
            if op != 1:
                return None
            param = None
            if e[pos + 1] == 0 and e[pos] < 0x20:      # a string never starts with a control byte
                param, pos = e[pos], pos + 2
            end = pos
            while e[end:end + 2] != b"\0\0":
                end += 2
                if end >= len(e):
                    return None
            lines.append((param, e[pos:end]))
            pos = end + 2
        entries.append(lines)
    return entries


def text_leaf(entries) -> bytes:
    body, offs, pos = bytearray(), [], 2 * len(entries)
    for lines in entries:
        offs.append(pos)
        e = bytearray()
        for param, s in lines:
            assert len(s) % 2 == 0 and b"\0\0" not in s
            e += b"\x01\x00" + (struct.pack("<H", param) if param is not None else b"") + s + b"\0\0"
        if lines:
            e += b"\0\0"
        body += e
        pos += len(e)
    if pos > 0xFFFF:
        raise ValueError(f"text leaf too large: {pos} bytes")
    return b"".join(struct.pack("<H", o) for o in offs) + bytes(body)
```

```python
A_COUNT, B_COUNT = 68, 48
A_WIDTH, B_WIDTH = 32, 28
BLANK = "　"


def _leaves(node):
    if isinstance(node, bytes):
        yield node
    else:
        for c in node[0]:
            yield from _leaves(c)


def _text_leaves(sub: bytes) -> dict:
    """{'A': entries, 'B': entries} for the two known leaves."""
    found = {}
    for leaf in _leaves(parse(sub)):
        entries = text_entries(leaf)
        if entries and len(entries) in (A_COUNT, B_COUNT):
            found["A" if len(entries) == A_COUNT else "B"] = entries
    return found


def has_text(sub: bytes) -> bool:
    return len(_text_leaves(sub)) == 2


def extract(sub: bytes):
    leaves = _text_leaves(sub)
    notebook = [{"id": f"TOWN/A/{i:02d}", "jp": [script.decode(s) for _, s in lines], "en": "",
                 "width": A_WIDTH, "wrap": True}
                for i, lines in enumerate(leaves["A"])]
    ui = [{"id": f"TOWN/B/{i:02d}/{j:02d}", "jp": script.decode(s), "en": "", "width": B_WIDTH}
          for i, lines in enumerate(leaves["B"]) for j, (_, s) in enumerate(lines)]
    return notebook, ui


def _article_lines(e: dict) -> list[str]:
    if not e["en"]:
        return list(e["jp"])
    out = []
    for para in e["en"].split("\n"):
        out += script.wrap(para, A_WIDTH) if para.strip() else [BLANK]
    return out


def _rebuild_leaf(node, old_entries, new_entries):
    """Replace, in place in the tree, the leaf whose entries equal old_entries."""
    if isinstance(node, bytes):
        return node
    children, tail = node
    for i, c in enumerate(children):
        if isinstance(c, bytes):
            if text_entries(c) == old_entries:
                children[i] = text_leaf(new_entries)
        else:
            _rebuild_leaf(c, old_entries, new_entries)
    return node


def insert(sub: bytes, notebook: list, ui: list) -> bytes:
    leaves = _text_leaves(sub)
    tree = parse(sub)
    by_id = {e["id"]: e for e in notebook}
    new_a = []
    for i, lines in enumerate(leaves["A"]):
        e = by_id.pop(f"TOWN/A/{i:02d}", None)
        if e is None:
            new_a.append(lines)
            continue
        new_a.append([(None, script.encode(l)) for l in _article_lines(e)])
    by_id.update({e["id"]: e for e in ui})
    new_b = []
    for i, lines in enumerate(leaves["B"]):
        out = []
        for j, (param, s) in enumerate(lines):
            e = by_id.pop(f"TOWN/B/{i:02d}/{j:02d}", None)
            if e and e["en"]:
                s = script.encode(e["en"])
                if len(s) > B_WIDTH:
                    raise ValueError(f"{e['id']}: {len(s)} bytes > {B_WIDTH}")
            out.append((param, s))
        new_b.append(out)
    if by_id:
        raise ValueError(f"no such TOWN text: {sorted(by_id)[:5]}")
    _rebuild_leaf(tree, leaves["A"], new_a)
    _rebuild_leaf(tree, leaves["B"], new_b)
    return rebuild(tree)
```

Note: `_rebuild_leaf` mutates the tree; identity insert is identical because `text_leaf(text_entries(leaf)) == leaf` (Task 2) and `rebuild` pads to the same 4-byte sizes.
